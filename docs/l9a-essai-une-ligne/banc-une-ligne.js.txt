// L9a (15/11) — LE SÉQUENCEUR S'ÉTIRE JUSQU'À NE LAISSER QU'UNE LIGNE D'ACCORDS À LA GRILLE. L'utilisateur : « Peux-tu faire en sorte que je puisse étirer le séquenceur au-dessus de la barre
// d'outils ? il y aura donc une seule ligne visible si je suis en plein écran pour le séquenceur » ; réponse à « une seule ligne de la grille d'accords ? » : « oui ok pour une ligne au-dessus ».
// Avant, la poignée s'arrêtait à 40 px de grille : il n'y restait que la rangée de réglages collée, aucun accord.
//
// Ce que le banc éprouve, à la souris (vrai glissé de la poignée du haut du séquenceur, `#sq-poignee`) :
//   A. étiré au plus haut : la grille n'a plus qu'UNE ligne, celle de l'accord choisi (quatre accords entiers, aucun de la ligne voisine) ; la rangée de réglages n'est plus collée ;
//      le séquenceur est plus haut qu'avant ; la page ne défile pas, la barre de lecture reste à l'écran ;
//   B. ▲ ▼ (ligne suivante) : la grille suit, toujours une seule ligne, celle que le séquenceur montre ;
//   C. redescendre la poignée (puis le double-clic) rend la grille et la rangée de réglages collée ;
//   D. le téléphone n'a pas bougé (aucune classe, rien de collé en moins) ; un écran plus bas ne casse rien.
const { chromium } = require('playwright');
const { check, exiger, plan, bilan } = require('./_harness')('séquenceur étiré : une ligne d\'accords reste à la grille (L9a)');
const { estBruitReseau } = require('./_harness');
const V = require('./_verre_commun');
const BASE = process.env.HARMOHUB_URL || 'http://localhost:8934';

plan(17);

const mk = (root, quality) => ({ root, quality, beats: 4, inversion: 0, drop: 'none', octave: 3, bass: null, playStyle: 'held' });
const SECTIONS = [
    { title: 'Couplet', chords: [mk('C', 'maj7'), mk('A', 'min7'), mk('D', 'min7'), mk('G', '7'), mk('E', 'min7'), mk('F', 'maj7'), mk('G', '7'), mk('C', 'maj7')] },
    { title: 'Refrain', chords: [mk('F', 'maj7'), mk('E', 'min7'), mk('D', 'min7'), mk('C', 'maj7')] },
];

async function ouvrir(nav, contexte, erreurs, nom) {
    const ctx = await nav.newContext(contexte);
    await ctx.addInitScript((sections) => {
        if (sessionStorage.getItem('banc-l9a')) return;
        sessionStorage.setItem('banc-l9a', '1');
        localStorage.setItem('harmohubSongs', JSON.stringify([{ id: 's1', name: 'Essai', bpm: 120, timeSig: '4/4', groove: 'straight', savedAt: Date.now(), sections }]));
        localStorage.setItem('harmohubCurrentSongId', 's1');
        localStorage.setItem('myProgression', JSON.stringify({ sections }));
        localStorage.setItem('harmohubSequenceurOuvert', '1');
        localStorage.setItem('harmohubSequenceurVue', 'notes');   // la vue Piano a des dizaines de lignes : de quoi étirer
        sessionStorage.setItem('cc_accueil_vu', '1');
    }, SECTIONS);
    await ctx.route(/gstatic\.com|googleapis\.com|tonejs\.github\.io/, r => r.abort());
    const p = await ctx.newPage();
    p.on('pageerror', e => erreurs.push(nom + e.message));
    p.on('console', m => { if (m.type() === 'error' && !estBruitReseau(m.text()) && !V.BRUIT.test(m.text())) erreurs.push(nom + 'console: ' + m.text()); });
    await p.goto(`${BASE}/index.html?nocache=` + Date.now(), { waitUntil: 'load' });
    await p.waitForFunction(() => !!window.app);
    await p.waitForTimeout(1000);
    return { p, ctx };
}

// Ce que la grille montre : la fenêtre de la grille, ses cases ENTIÈREMENT visibles (par numéro d'accord), la rangée de réglages, la page, la barre de lecture.
const etat = (p) => p.evaluate(() => {
    const aire = document.querySelector('.history-section').getBoundingClientRect();
    const entieres = [...document.querySelectorAll('#progression-sections .grid-cell[data-index]')].filter(c => { const r = c.getBoundingClientRect(); return r.top >= aire.top - 1 && r.bottom <= aire.bottom + 1 && r.width > 0; })
        .map(c => c.closest('.chord-grid').dataset.section + ':' + c.dataset.index);
    const sq = document.getElementById('sequenceur').getBoundingClientRect(), bl = document.getElementById('barre-lecture').getBoundingClientRect();
    return { aireH: Math.round(aire.height), entieres: entieres.join(), classe: document.body.classList.contains('grille-une-ligne'), collee: getComputedStyle(document.querySelector('.grid-head-sticky')).position, bandeau: getComputedStyle(document.getElementById('add-section')).display,
        sqH: Math.round(sq.height), sqTop: Math.round(sq.top), aireBas: Math.round(aire.bottom), pageH: document.documentElement.scrollHeight, vh: innerHeight, barreBas: Math.round(bl.bottom) };
});
const tirer = async (p, versY, pas = 25) => {
    const h = await p.locator('#sq-poignee').boundingBox();
    const x = h.x + h.width / 2, y0 = h.y + h.height / 2;
    await p.mouse.move(x, y0); await p.mouse.down();
    const n = Math.max(2, Math.round(Math.abs(versY - y0) / pas));
    for (let i = 1; i <= n; i++) { await p.mouse.move(x, y0 + (versY - y0) * i / n); await p.waitForTimeout(15); }
    await p.mouse.up(); await p.waitForTimeout(500);
};

(async () => {
    const nav = await chromium.launch();
    const erreurs = [];
    try {
        // ===== A. ÉTIRÉ AU PLUS HAUT =================================================================================================
        console.log('=== A. La poignée du haut jusqu\'en haut : une ligne d\'accords reste ===');
        let { p, ctx } = await ouvrir(nav, { viewport: { width: 1440, height: 900 }, deviceScaleFactor: 1 }, erreurs, 'ordinateur : ');
        await p.click('#progression-sections .chord-grid[data-section="0"] .grid-cell[data-index="1"]'); await p.waitForTimeout(500);
        const avant = await etat(p);
        check(!avant.classe && avant.collee === 'sticky' && avant.aireH > 170, `au départ la grille a de la place et sa rangée de réglages est collée (${avant.aireH} px, ${avant.collee})`);
        await tirer(p, 40);
        const haut = await etat(p);
        check(haut.classe && haut.collee === 'static' && haut.bandeau === 'none' && avant.bandeau !== 'none', `étiré au plus haut, la rangée de réglages n'est plus collée et le bandeau « Ajouter section » s'efface (classe ${haut.classe}, ${haut.collee}, bandeau ${haut.bandeau})`);
        check(haut.aireH >= 85 && haut.aireH <= 110, `la grille n'a plus que la hauteur d'UNE ligne (${haut.aireH} px, 36 px avant)`);
        check(haut.entieres === '0:0,0:1,0:2,0:3', `elle montre la ligne de l'accord choisi, quatre accords entiers et aucun d'une autre ligne (${haut.entieres})`);
        check(haut.sqH > avant.sqH + 60 && haut.sqTop >= haut.aireBas, `le séquenceur est plus haut (${avant.sqH} → ${haut.sqH} px) et ne recouvre pas la ligne (haut à ${haut.sqTop}, ligne jusqu'à ${haut.aireBas})`);
        check(haut.pageH === haut.vh && haut.barreBas === haut.vh, `la page ne défile pas et la barre de lecture reste à l'écran (page ${haut.pageH}/${haut.vh}, barre jusqu'à ${haut.barreBas})`);
        const encore = await p.evaluate(() => document.getElementById('sq-poignee').getBoundingClientRect().top);
        await tirer(p, 10);
        check(Math.abs((await p.evaluate(() => document.getElementById('sq-poignee').getBoundingClientRect().top)) - encore) <= 2, 'tirer encore plus haut ne va pas plus loin : la ligne ne disparaît jamais');

        // ===== B. LA LIGNE SUIT ======================================================================================================
        console.log('\n=== B. ▲ ▼ : la grille suit la ligne montrée ===');
        await p.click('#sq-ligne-suiv'); await p.waitForTimeout(600);
        const suiv = await etat(p);
        check(suiv.entieres === '0:4,0:5,0:6,0:7' && suiv.classe, `▼ : la ligne suivante, seule à la grille (${suiv.entieres})`);
        await p.click('#sq-ligne-suiv'); await p.waitForTimeout(600);
        const refrain = await etat(p);
        check(refrain.entieres === '1:0,1:1,1:2,1:3', `▼ encore : la ligne du Refrain (${refrain.entieres})`);
        await p.click('#sq-ligne-prec'); await p.waitForTimeout(600);
        check((await etat(p)).entieres === '0:4,0:5,0:6,0:7', '▲ revient en arrière');
        await p.click('#progression-sections .chord-grid[data-section="0"] .grid-cell[data-index="5"]'); await p.waitForTimeout(500);
        const clic = await etat(p);
        check(clic.entieres === '0:4,0:5,0:6,0:7' && clic.classe, `cliquer un accord de la ligne la garde (${clic.entieres})`);

        // ===== C. REDESCENDRE ========================================================================================================
        console.log('\n=== C. Redescendre la poignée rend la grille ===');
        const h = await p.locator('#sq-poignee').boundingBox();
        await tirer(p, h.y + h.height / 2 + 160);
        const bas = await etat(p);
        check(!bas.classe && bas.collee === 'sticky' && bas.aireH > 170, `la grille a repris de la place, la rangée de réglages est de nouveau collée (${bas.aireH} px, ${bas.collee})`);
        await tirer(p, 40);
        check((await etat(p)).classe, 'tirée de nouveau tout en haut : une ligne');
        await p.dblclick('#sq-poignee'); await p.waitForTimeout(500);
        const dbl = await etat(p);
        check(!dbl.classe && dbl.collee === 'sticky', `le double-clic sur la poignée rend tout (${dbl.collee}, ${dbl.aireH} px)`);
        await ctx.close();

        // ===== D. UN ÉCRAN BAS, LE TÉLÉPHONE ==========================================================================================
        console.log('\n=== D. Écran bas, téléphone ===');
        ({ p, ctx } = await ouvrir(nav, { viewport: { width: 1280, height: 700 }, deviceScaleFactor: 1 }, erreurs, 'bas : '));
        await p.click('#progression-sections .chord-grid[data-section="0"] .grid-cell[data-index="1"]'); await p.waitForTimeout(500);
        await tirer(p, 30);
        const bas7 = await etat(p);
        check(bas7.pageH === bas7.vh && bas7.barreBas === bas7.vh && bas7.entieres === '0:0,0:1,0:2,0:3', `à 700 px de haut aussi : une ligne, la barre de lecture à l'écran (${bas7.entieres}, page ${bas7.pageH}/${bas7.vh})`);
        await ctx.close();
        ({ p, ctx } = await ouvrir(nav, { viewport: { width: 390, height: 844 }, hasTouch: true, isMobile: true, deviceScaleFactor: 1 }, erreurs, 'téléphone : '));
        const tel = await p.evaluate(() => ({ classe: document.body.classList.contains('grille-une-ligne'), collee: getComputedStyle(document.querySelector('.grid-head-sticky')).position }));
        check(!tel.classe, `le téléphone n'a jamais cette classe (${tel.classe})`);
        await ctx.close();
    } catch (e) {
        console.log('FATAL ' + e.message);
        process.exitCode = 1;
    }
    check(erreurs.length === 0, `aucune erreur de page ni de console${erreurs.length ? ' : ' + erreurs.slice(0, 3).join(' | ') : ''}`);
    await nav.close();
    bilan();
})();
